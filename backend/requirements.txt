# EcoTrack/backend/requirements.txt
# Install everything with:  pip install -r requirements.txt

Flask==3.1.0
Flask-Cors==5.0.0
firebase-admin==6.6.0
python-dotenv==1.0.1

# firebase-admin only requires google-cloud-firestore>=2.19.0 with no upper
# bound, so a fresh install (every Vercel build - there is no committed lock
# file) is free to resolve whatever is newest on PyPI at build time. Pinned
# here after a newer resolved version broke every Firestore call in
# production with "400 Invalid database id (default)" while this exact
# version, tested directly against the same production database, did not -
# see the incident notes in git history around this line for the full
# traceback. Bump deliberately, not by drift.
google-cloud-firestore==2.28.0

# Powers the EcoTrack Assistant (routes/assistant.py).
# Google AI Studio has a free tier that needs no credit card, which is why the
# assistant uses Gemini rather than a paid API.
# The app still runs without this installed - the assistant routes simply
# report that the feature is unavailable instead of crashing the server.
google-genai>=2.0.0

# gunicorn is the production web server used by Render.
# It is not needed on Windows for local development, but keeping it here
# means Render installs it automatically during deployment.
gunicorn==23.0.0

# The Razorpay donation routes (routes/payments.py) call the Razorpay REST API
# directly over HTTPS and verify the signature with the standard-library hmac
# module - no razorpay SDK needed. requests already ships as a firebase-admin
# dependency; pinning it here makes that explicit since we import it directly.
requests>=2.31.0

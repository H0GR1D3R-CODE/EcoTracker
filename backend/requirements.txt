# firebase-admin only requires lower bounds (google-cloud-firestore>=2.19.0,
# etc.) with no upper bound, and there is no committed lock file, so a fresh
# Vercel build is free to resolve whatever is newest on PyPI at build time.
# Pinned to this exact, verified-working constellation after a newer
# transitive resolution broke every Firestore call in production with
# "400 Invalid database id (default)" - traced to a double percent-encoded
# database path segment (projects/.../databases/%2528default%2529 instead of
# .../databases/(default)) coming from the google-api-core/grpc transport
# layer, not google-cloud-firestore itself. Bump this whole group
# deliberately together, not by drift - a single package left unpinned can
# silently take the others' compatibility guarantees with it.
google-cloud-firestore==2.28.0
google-api-core==2.33.0
google-auth==2.56.2
grpcio==1.83.0

# Powers the EcoTrack Assistant (routes/assistant.py).
# Groq's free tier needs no credit card, and its custom inference hardware is
# why the assistant moved here from Gemini - see config.py's GROQ_API_KEY
# comment for the fuller reasoning.
# The app still runs without this installed - the assistant routes simply
# report that the feature is unavailable instead of crashing the server.
groq==1.7.0

# gunicorn is the production web server used by Render.
# It is not needed on Windows for local development, but keeping it here
# means Render installs it automatically during deployment.
gunicorn==23.0.0

# The Razorpay donation routes (routes/payments.py) call the Razorpay REST API
# directly over HTTPS and verify the signature with the standard-library hmac
# module - no razorpay SDK needed. requests already ships as a firebase-admin
# dependency; pinning it here makes that explicit since we import it directly.
requests>=2.31.0
